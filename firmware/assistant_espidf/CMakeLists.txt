cmake_minimum_required(VERSION 3.19)
set(ASSISTANT_BOARD "waveshare-amoled-1.8" CACHE STRING "Assistant hardware profile")
include(${CMAKE_CURRENT_LIST_DIR}/board_profiles.cmake)
if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL ASSISTANT_BOARD_TARGET)
    message(FATAL_ERROR "Requested chip target differs from the selected board profile")
endif()
set(IDF_TARGET "${ASSISTANT_BOARD_TARGET}" CACHE STRING "Chip target of the selected board")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig_target_${ASSISTANT_BOARD_TARGET}.defaults")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig_board.defaults")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig_board.defaults")
elseif(EXISTS "${CMAKE_CURRENT_LIST_DIR}/boards/${ASSISTANT_BOARD}/sdkconfig.defaults")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/boards/${ASSISTANT_BOARD}/sdkconfig.defaults")
endif()
file(READ ${CMAKE_CURRENT_LIST_DIR}/main/version.txt PROJECT_VER)
string(STRIP "${PROJECT_VER}" PROJECT_VER)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(${ASSISTANT_PROJECT_NAME})
